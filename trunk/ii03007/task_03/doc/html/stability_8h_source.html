18// Решение кубического уравнения (Кардано / тригонометрический метод, в double).
19inline CubicRoots cubicRoots(double a1, double a2, double a3) {
20 CubicRoots r{};
21 const double PI = 3.14159265358979323846;
22
23 // Уравнение вида z^3 + p*z^2 + q*z + rr = 0
24 double p = -a1;
25 double q = -a2;
26 double rr = -a3;
27
28 // Приведённый кубик t^3 + A*t + B = 0, замена z = t - p/3
29 double A = q - p * p / 3.0;
30
31 if (double B = 2.0 * p * p * p / 27.0 - p * q / 3.0 + rr; B * B / 4.0 + A * A * A / 27.0 > 0.0) {
32 // Один вещественный корень + комплексно-сопряжённая пара
33 double s = std::sqrt(B * B / 4.0 + A * A * A / 27.0);
34 double t1 = std::cbrt(-B / 2.0 + s) + std::cbrt(-B / 2.0 - s);
35 r.re[0] = t1 - p / 3.0;
36 r.im[0] = 0.0;
37 double dq = 3.0 * t1 * t1 + 4.0 * A; // положительно при B * B / 4.0 + A * A * A / 27.0 > 0
38 r.re[1] = -t1 / 2.0 - p / 3.0;
39 r.im[1] = 0.5 * std::sqrt(dq > 0.0 ? dq : 0.0);
40 r.re[2] = r.re[1];
41 r.im[2] = -r.im[1];
42 } else {
43 // Три вещественных корня (B * B / 4.0 + A * A * A / 27.0 = 0 - кратные)
44 if (A >= 0.0) {
45 // a1 = a2 = a3 = 0: тройной корень t = 0
46 r.re[0] = -p / 3.0;
47 r.re[1] = -p / 3.0;
48 r.re[2] = -p / 3.0;
49 r.im[0] = 0.0;
50 r.im[1] = 0.0;
51 r.im[2] = 0.0;
52 } else {
53 double m = 2.0 * std::sqrt(-A / 3.0);
54 double arg = 3.0 * B / (2.0 * A) * std::sqrt(-3.0 / A);
55 if (arg > 1.0) arg = 1.0;
56 if (arg < -1.0) arg = -1.0;
57 double phi = std::acos(arg) / 3.0;
58 for (int k = 0; k < 3; ++k) {
59 r.re[k] = m * std::cos(phi - 2.0 * PI * k / 3.0) - p / 3.0;
60 r.im[k] = 0.0;
61 }
62 }
63 }
64
65 for (int k = 0; k < 3; ++k)
66 r.mod[k] = std::sqrt(r.re[k] * r.re[k] + r.im[k] * r.im[k]);
67
68 // Округляем до 9 знаков перед сравнением: корень на окружности
69 // из-за округления может выйти 1 +/- 1e-16
70 r.stable = true;
71 for (int k = 0; k < 3; ++k) {
72 double m = std::round(r.mod[k] * 1e9) / 1e9;
73 if (m >= 1.0) r.stable = false;
74 }
75 return r;
76}
