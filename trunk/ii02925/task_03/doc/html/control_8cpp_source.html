59double PIDController::update(double error) {
60 const double derivative = first_ ? 0.0 : (error - previousError_) / dt_;
61 const double candidateIntegral = integral_ + error * dt_;
62 const double raw = kp_*error + ki_*candidateIntegral + kd_*derivative;
63 const double limited = std::clamp(raw, min_, max_);
64 // Conditional integration: do not grow the integral further into saturation.
65 if (const bool pushesHigh = raw > max_ && error > 0.0;
66 !pushesHigh && !(raw < min_ && error < 0.0)) {
67 integral_ = candidateIntegral;
68 }
69 previousError_ = error;
70 first_ = false;
71 return limited;
72}

102 for (int step = 0; step < maximumSteps; ++step) {
103 const double error = setpoint - plant.output();
104 const double control = pid.update(error);
105 const double output = plant.update(control);
