45double PIDController::update(double error) {
46 const double derivative = first_ ? 0.0 : (error - previousError_) / dt_;
47 const double candidateIntegral = integral_ + error * dt_;
48 const double raw = kp_*error + ki_*candidateIntegral + kd_*derivative;
49 const double limited = std::clamp(raw, min_, max_);
50 // Conditional integration: do not grow the integral further into saturation.
51 const bool pushesHigh = raw > max_ && error > 0.0;
52 const bool pushesLow = raw < min_ && error < 0.0;
53 if (!pushesHigh && !pushesLow) integral_ = candidateIntegral;
54 previousError_ = error;
55 first_ = false;
56 return limited;
57}
