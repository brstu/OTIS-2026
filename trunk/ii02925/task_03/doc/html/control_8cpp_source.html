44double PIDController::update(double error) {
45 const double derivative = first_ ? 0.0 : (error - previousError_) / dt_;
46 const double candidateIntegral = integral_ + error * dt_;
47 const double raw = kp_*error + ki_*candidateIntegral + kd_*derivative;
48 const double limited = std::clamp(raw, min_, max_);
49 // Conditional integration: do not grow the integral further into saturation.
50 const bool pushesHigh = raw > max_ && error > 0.0;
51 const bool pushesLow = raw < min_ && error < 0.0;
52 if (!pushesHigh && !pushesLow) integral_ = candidateIntegral;
53 previousError_ = error;
54 first_ = false;
55 return limited;
56}
